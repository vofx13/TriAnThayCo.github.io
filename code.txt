            function createNumber(number, isFirst = false) {
                const pattern = numberPatterns[number];
                if (!pattern) return;
                
                const blockSize = 6;
                const gap = 2;
                const rows = pattern.length;
                const cols = pattern[0].length;
                
                const totalWidth = cols * (blockSize + gap) - gap;
                const totalHeight = rows * (blockSize + gap) - gap;
                
                const startX = (countdownArea.offsetWidth - totalWidth) / 2;
                const startY = (countdownArea.offsetHeight - totalHeight) / 2;
                
                // Clear existing blocks
                currentBlocks.forEach(block => {
                    if (block.parentNode === countdownArea) {
                        countdownArea.removeChild(block);
                    }
                });
                currentBlocks = [];
                
                for (let row = 0; row < rows; row++) {
                    for (let col = 0; col < cols; col++) {
                        if (pattern[row][col] === 1) {
                            const block = document.createElement('div');
                            block.className = 'block';
                            block.style.width = `${blockSize}px`;
                            block.style.height = `${blockSize}px`;
                            
                            const finalX = startX + col * (blockSize + gap);
                            const finalY = startY + row * (blockSize + gap);
                            
                            // For first number, start from random positions
                            let startPosX, startPosY;
                            if (isFirst) {
                                startPosX = Math.random() * countdownArea.offsetWidth;
                                startPosY = Math.random() * countdownArea.offsetHeight;
                            } else {
                                startPosX = finalX;
                                startPosY = finalY;
                                block.style.opacity = '0';
                            }
                            
                            block.style.left = `${startPosX}px`;
                            block.style.top = `${startPosY}px`;
                            
                            const startTime = Date.now();
                            const duration = 150 + Math.random() * 100; // Faster animation for mobile
                            
                            function animateBlock() {
                                const elapsed = Date.now() - startTime;
                                const progress = Math.min(elapsed / duration, 1);
                                
                                // Easing function for smooth motion
                                const ease = 1 - Math.pow(1 - progress, 3);
                                
                                const currentX = startPosX + (finalX - startPosX) * ease;
                                const currentY = startPosY + (finalY - startPosY) * ease;
                                
                                block.style.left = `${currentX}px`;
                                block.style.top = `${currentY}px`;
                                block.style.opacity = progress.toString();
                                
                                if (progress < 1) {
                                    requestAnimationFrame(animateBlock);
                                }
                            }
                            
                            // Minimal stagger for faster effect
                            setTimeout(() => {
                                requestAnimationFrame(animateBlock);
                            }, Math.random() * 20); // Reduced stagger for mobile
                            
                            countdownArea.appendChild(block);
                            currentBlocks.push(block);
                        }
                    }
                }
            }
            
            function createMessage(text, duration = 1500) {
                // Clear existing blocks
                currentBlocks.forEach(block => {
                    if (block.parentNode === messageArea) {
                        messageArea.removeChild(block);
                    }
                });
                currentBlocks = [];
                
                const blockSize = 4;
                const gap = 1;
                const letterSpacing = 8;
                
                // Calculate total width
                let totalWidth = 0;
                for (let i = 0; i < text.length; i++) {
                    const letter = text[i];
                    if (letter === ' ') {
                        totalWidth += 15;
                    } else {
                        totalWidth += letterPatterns[letter][0].length * (blockSize + gap) - gap;
                        if (i < text.length - 1) totalWidth += letterSpacing;
                    }
                }
                
                const startX = (messageArea.offsetWidth - totalWidth) / 2;
                const startY = (messageArea.offsetHeight - 8 * (blockSize + gap) - gap) / 2;
                
                let currentX = startX;
                
                // Calculate timing for specified duration assembly
                const totalLetters = text.replace(/ /g, '').length;
                const timePerLetter = duration / totalLetters;
                
                // Create blocks for each letter with precise timing
                function createLetter(index, startTime) {
                    if (index >= text.length) {
                        // Schedule next message after a delay
                        setTimeout(() => {
                            currentMessageIndex++;
                            if (currentMessageIndex < messages.length) {
                                explodeCurrentMessage();
                            } else {
                                // After last message, show the envelope
                                setTimeout(showEnvelope, 300); // Reduced delay for mobile
                            }
                        }, 300); // Reduced delay before next message
                        return;
                    }
                    
                    const letter = text[index];
                    
                    if (letter === ' ') {
                        currentX += 15;
                        createLetter(index + 1, startTime);
                        return;
                    }
                    
                    const pattern = letterPatterns[letter];
                    if (!pattern) {
                        createLetter(index + 1, startTime);
                        return;
                    }
                    
                    const rows = pattern.length;
                    const cols = pattern[0].length;
                    const letterWidth = cols * (blockSize + gap) - gap;
                    
                    const letterBlocks = [];
                    
                    for (let row = 0; row < rows; row++) {
                        for (let col = 0; col < cols; col++) {
                            if (pattern[row][col] === 1) {
                                const block = document.createElement('div');
                                block.className = 'block';
                                block.style.width = `${blockSize}px`;
                                block.style.height = `${blockSize}px`;
                                
                                const finalX = currentX + col * (blockSize + gap);
                                const finalY = startY + row * (blockSize + gap);
                                
                                // Start from random positions for animation
                                const startPosX = Math.random() * messageArea.offsetWidth;
                                const startPosY = Math.random() * messageArea.offsetHeight;
                                
                                block.style.left = `${startPosX}px`;
                                block.style.top = `${startPosY}px`;
                                block.style.opacity = '0';
                                
                                messageArea.appendChild(block);
                                currentBlocks.push(block);
                                letterBlocks.push({
                                    block,
                                    startPosX,
                                    startPosY,
                                    finalX,
                                    finalY
                                });
                            }
                        }
                    }
                    
                    // Animate this letter's blocks
                    const targetDuration = timePerLetter * (letterBlocks.length / 10);
                    animateLetterBlocks(letterBlocks, 0, startTime, targetDuration, () => {
                        currentX += letterWidth + letterSpacing;
                        createLetter(index + 1, startTime + targetDuration);
                    });
                }
                
                // Start creating letters
                createLetter(0, Date.now());
            }
            
            function animateLetterBlocks(letterBlocks, index, startTime, targetDuration, callback) {
                if (index >= letterBlocks.length) {
                    callback();
                    return;
                }
                
                const blockData = letterBlocks[index];
                const { block, startPosX, startPosY, finalX, finalY } = blockData;
                
                // Faster animation speeds for mobile
                const duration = 30 + Math.random() * 40; // Very fast animation
                
                function animateBlock() {
                    const elapsed = Date.now() - startTime;
                    const progress = Math.min(elapsed / duration, 1);
                    
                    // Easing function for smooth motion
                    const ease = 1 - Math.pow(1 - progress, 3);
                    
                    const currentX = startPosX + (finalX - startPosX) * ease;
                    const currentY = startPosY + (finalY - startPosY) * ease;
                    
                    block.style.left = `${currentX}px`;
                    block.style.top = `${currentY}px`;
                    block.style.opacity = progress.toString();
                    
                    if (progress < 1) {
                        requestAnimationFrame(animateBlock);
                    } else {
                        // Move to next block immediately
                        animateLetterBlocks(letterBlocks, index + 1, startTime, targetDuration, callback);
                    }
                }
                
                // Start animation immediately
                requestAnimationFrame(animateBlock);
            }
            
            function explodeCurrentMessage() {
                // Create particles during explosion
                createParticles();
                
                currentBlocks.forEach((block, index) => {
                    const startX = parseFloat(block.style.left);
                    const startY = parseFloat(block.style.top);
                    
                    const angle = Math.random() * Math.PI * 2;
                    const distance = 30 + Math.random() * 50;
                    const targetX = startX + Math.cos(angle) * distance;
                    const targetY = startY + Math.sin(angle) * distance;
                    
                    const startTime = Date.now();
                    const duration = 100 + Math.random() * 80; // Faster explosion for mobile
                    
                    function animateExplosion() {
                        const elapsed = Date.now() - startTime;
                        const progress = Math.min(elapsed / duration, 1);
                        
                        // Easing function for explosion
                        const ease = Math.pow(progress, 0.7);
                        
                        const currentX = startX + (targetX - startX) * ease;
                        const currentY = startY + (targetY - startY) * ease;
                        const rotation = progress * 360;
                        const opacity = 1 - progress;
                        
                        block.style.left = `${currentX}px`;
                        block.style.top = `${currentY}px`;
                        block.style.transform = `rotate(${rotation}deg)`;
                        block.style.opacity = opacity.toString();
                        
                        if (progress < 1) {
                            requestAnimationFrame(animateExplosion);
                        } else {
                            // Remove block after explosion
                            if (block.parentNode === messageArea) {
                                block.parentNode.removeChild(block);
                            }
                        }
                    }
                    
                    // Minimal stagger for faster explosions
                    setTimeout(() => {
                        requestAnimationFrame(animateExplosion);
                    }, index * 2); // Reduced stagger for mobile
                });
                
                // After explosion, show next message
                setTimeout(() => {
                    showNextMessage();
                }, 200); // Reduced delay for mobile
            }
            
            function showNextMessage() {
                if (currentMessageIndex < messages.length) {
                    const message = messages[currentMessageIndex];
                    const duration = messageDurations[currentMessageIndex];
                    createMessage(message, duration);
                }
            }

            }
            
            function createParticles() {
                for (let i = 0; i < 10; i++) { // Fewer particles for mobile
                    const particle = document.createElement('div');
                    particle.className = 'particle';
                    
                    // Position particles around the center
                    const centerX = messageArea.offsetWidth / 2;
                    const centerY = messageArea.offsetHeight / 2;
                    
                    particle.style.left = `${centerX}px`;
                    particle.style.top = `${centerY}px`;
                    
                    document.querySelector('.container').appendChild(particle);
                    
                    // Animate particles
                    const angle = Math.random() * Math.PI * 2;
                    const distance = 50 + Math.random() * 80;
                    const targetX = centerX + Math.cos(angle) * distance;
                    const targetY = centerY + Math.sin(angle) * distance;
                    
                    const startTime = Date.now();
                    const duration = 150 + Math.random() * 100; // Faster particles for mobile
                    
                    function animateParticle() {
                        const elapsed = Date.now() - startTime;
                        const progress = Math.min(elapsed / duration, 1);
                        
                        const currentX = centerX + (targetX - centerX) * progress;
                        const currentY = centerY + (targetY - centerY) * progress;
                        const opacity = 1 - progress;
                        
                        particle.style.left = `${currentX}px`;
                        particle.style.top = `${currentY}px`;
                        particle.style.opacity = opacity.toString();
                        
                        if (progress < 1) {
                            requestAnimationFrame(animateParticle);
                        } else {
                            // Remove particle after animation
                            if (particle.parentNode) {
                                particle.parentNode.removeChild(particle);
                            }
                        }
                    }
                    
                    requestAnimationFrame(animateParticle);
                }
            }
            
            function createConfetti() {
                const colors = ['#e6d2aa', '#ffebcd', '#f5deb3', '#fff8dc', '#faf0e6'];
                
                for (let i = 0; i < 40; i++) { // Fewer confetti for mobile
                    const confetti = document.createElement('div');
                    confetti.className = 'confetti';
                    
                    // Random color
                    confetti.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];
                    
                    // Random position at top of screen
                    confetti.style.left = `${Math.random() * 100}vw`;
                    confetti.style.top = '-20px';
                    
                    // Random rotation
                    confetti.style.transform = `rotate(${Math.random() * 360}deg)`;
                    
                    document.querySelector('.container').appendChild(confetti);
                    
                    // Animate confetti falling
                    const startTime = Date.now();
                    const duration = 1000 + Math.random() * 800;
                    const endY = window.innerHeight + 20;
                    
                    function animateConfetti() {
                        const elapsed = Date.now() - startTime;
                        const progress = Math.min(elapsed / duration, 1);
                        
                        const currentY = -20 + (endY + 20) * progress;
                        const currentX = parseFloat(confetti.style.left) + Math.sin(progress * 10) * 20;

                                      progress > 0.9 ? 1 - (progress - 0.9) * 10 : 1;
                        const rotation = parseFloat(confetti.style.transform.replace('rotate(', '').replace('deg)', '')) + 5;

                        confetti.style.top = `${currentY}px`;
                        confetti.style.transform = `rotate(${rotation}deg)`;
                        confetti.style.opacity = opacity.toString();
                        
                        if (progress < 1) {
                            requestAnimationFrame(animateConfetti);
                        } else {
                            // Remove confetti after animation
                            if (confetti.parentNode) {
                                confetti.parentNode.removeChild(confetti);
                            }
                        }
                    }
                    
                    // Minimal stagger
                    setTimeout(() => {
                        requestAnimationFrame(animateConfetti);
                    }, Math.random() * 200);
                }

            function startCountdown() {
                countdownValue = 3;
                createNumber(countdownValue, true);
                
                const countdownInterval = setInterval(() => {
                    if (countdownValue === 3) {
                        explodeBlocks();
                        setTimeout(() => {
                            countdownValue--;
                            createNumber(countdownValue);
                        }, 300); // Faster transitions for mobile
                    } else if (countdownValue === 2) {
                        setTimeout(() => {
                            explodeBlocks();
                        }, 600); // Faster transitions for mobile
                        
                        setTimeout(() => {
                            countdownValue--;
                            createNumber(countdownValue);
                        }, 900); // Faster transitions for mobile
                    } else if (countdownValue === 1) {
                        clearInterval(countdownInterval);
                        setTimeout(() => {
                            explodeBlocks();
                            // Show birthday message after final explosion
                            setTimeout(showBirthdayMessage, 200); // Faster transition for mobile
                        }, 400); // Faster transition for mobile
                    }
                }, 1000); // Faster countdown for mobile

            function explodeBlocks() {
                // Create particles during explosion
                createParticles();
                
                currentBlocks.forEach((block, index) => {
                    const startX = parseFloat(block.style.left);
                    const startY = parseFloat(block.style.top);
                    
                    const angle = Math.random() * Math.PI * 2;
                    const distance = 30 + Math.random() * 50;
                    const targetX = startX + Math.cos(angle) * distance;
                    const targetY = startY + Math.sin(angle) * distance;
                    
                    const startTime = Date.now();
                    const duration = 100 + Math.random() * 80; // Faster explosions for mobile
                    
                    function animateExplosion() {
                        const elapsed = Date.now() - startTime;
                        const progress = Math.min(elapsed / duration, 1);
                        
                        // Easing function for explosion
                        const ease = Math.pow(progress, 0.7);
                        
                        const currentX = startX + (targetX - startX) * ease;
                        const currentY = startY + (targetY - startY) * ease;
                        const rotation = progress * 360;
                        const opacity = 1 - progress;
                        
                        block.style.left = `${currentX}px`;
                        block.style.top = `${currentY}px`;
                        block.style.transform = `rotate(${rotation}deg)`;
                        block.style.opacity = opacity.toString();
                        
                        if (progress < 1) {
                            requestAnimationFrame(animateExplosion);
                        } else {
                            // Remove block after explosion
                            if (block.parentNode === countdownArea) {
                                block.parentNode.removeChild(block);
                            }
                        }
                    }
                    
                    // Minimal stagger for faster explosions
                    setTimeout(() => {
                        requestAnimationFrame(animateExplosion);
                    }, index * 2); // Reduced stagger for mobile
                });
            }

            function createNumber(number, isInitial = false) {
                // Clear existing blocks
                currentBlocks.forEach(block => {
                    if (block.parentNode === countdownArea) {
                        countdownArea.removeChild(block);
                    }
                });
                currentBlocks = [];
                
                const pattern = numberPatterns[number];

                const totalHeight = pattern.length * (blockSize + gap) - gap;
                
                const startX = (countdownArea.offsetWidth - totalWidth) / 2;
                const startY = (countdownArea.offsetHeight - totalHeight) / 2;
                
                // Create blocks for the number
                for (let row = 0; row < pattern.length; row++) {
                    for (let col = 0; col < pattern[row].length; col++) {

                            const block = document.createElement('div');
                            block.className = 'block';
                            
                            const finalX = startX + col * (blockSize + gap);
                            const finalY = startY + row * (blockSize + gap);

                                // For initial number, just place them directly
                                block.style.left = `${finalX}px`;
                                block.style.top = `${finalY}px`;
                                block.style.opacity = '1';
                            } else {
                                // For subsequent numbers, start from random positions
                                const startPosX = Math.random() * countdownArea.offsetWidth;
                                const startPosY = Math.random() * countdownArea.offsetHeight;
                                
                                block.style.left = `${startPosX}px`;
                                block.style.top = `${startPosY}px`;
                                block.style.opacity = '0';
                                
                                // Use requestAnimationFrame for smoother animation
                                const startTime = Date.now();
                                const duration = 200 + Math.random() * 100; // Faster animation
                                
                                function animateBlock() {
                                    const elapsed = Date.now() - startTime;
                                    const progress = Math.min(elapsed / duration, 1);
                                    
                                    // Easing function for smooth motion
                                    const ease = 1 - Math.pow(1 - progress, 3);
                                    
                                    const currentX = startPosX + (finalX - startPosX) * ease;
                                    const currentY = startPosY + (finalY - startPosY) * ease;
                                    
                                    block.style.left = `${currentX}px`;
                                    block.style.top = `${currentY}px`;
                                    block.style.opacity = progress.toString();
                                    
                                    if (progress < 1) {
                                        requestAnimationFrame(animateBlock);
                                    }
                                }
                                
                                // Minimal stagger for faster effect
                                setTimeout(() => {
                                    requestAnimationFrame(animateBlock);
                                }, Math.random() * 30); // Reduced stagger
                            }

                            currentBlocks.push(block);
                        }
                    }
                }

            function createMessage(text, duration = 2000) {
                // Clear existing blocks
                currentBlocks.forEach(block => {
                    if (block.parentNode === messageArea) {
                        messageArea.removeChild(block);
                    }
                });
                currentBlocks = [];
                
                const blockSize = 6;
                const gap = 2;
                const letterSpacing = 12;
                
                // Calculate total width
                let totalWidth = 0;
                for (let i = 0; i < text.length; i++) {
                    const letter = text[i];
                    if (letter === ' ') {
                        totalWidth += 25;
                    } else {
                        totalWidth += letterPatterns[letter][0].length * (blockSize + gap) - gap;
                        if (i < text.length - 1) totalWidth += letterSpacing;
                    }
                }
                
                const startX = (messageArea.offsetWidth - totalWidth) / 2;
                const startY = (messageArea.offsetHeight - 12 * (blockSize + gap) - gap) / 2;
                
                let currentX = startX;
                
                // Calculate timing for specified duration assembly
                const totalLetters = text.replace(/ /g, '').length;
                const timePerLetter = duration / totalLetters;
                
                // Create blocks for each letter with precise timing
                function createLetter(index, startTime) {
                    if (index >= text.length) {
                        // Schedule next message after a delay
                        setTimeout(() => {
                            currentMessageIndex++;
                            if (currentMessageIndex < messages.length) {
                                explodeCurrentMessage();
                            } else {
                                // After last message, show the envelope
                                setTimeout(showEnvelope, 500); // Reduced delay
                            }
                        }, 500); // Reduced delay before next message
                        return;
                    }
                    
                    const letter = text[index];
                    
                    if (letter === ' ') {
                        currentX += 25;
                        // Move to next letter immediately
                        createLetter(index + 1, startTime);
                        return;
                    }
                    
                    const pattern = letterPatterns[letter];
                    
                    // Create blocks for this letter
                    const letterBlocks = [];
                    for (let row = 0; row < pattern.length; row++) {
                        for (let col = 0; col < pattern[row].length; col++) {
                            if (pattern[row][col] === 1) {
                                const block = document.createElement('div');
                                block.className = 'block';
                                
                                const finalX = currentX + col * (blockSize + gap);
                                const finalY = startY + row * (blockSize + gap);
                                
                                // Start from random positions
                                const startPosX = Math.random() * messageArea.offsetWidth;
                                const startPosY = Math.random() * messageArea.offsetHeight;
                                
                                block.style.left = `${startPosX}px`;
                                block.style.top = `${startPosY}px`;
                                block.style.opacity = '0';
                                
                                letterBlocks.push({
                                    block: block,
                                    finalX: finalX,
                                    finalY: finalY,
                                    startPosX: startPosX,
                                    startPosY: startPosY
                                });
                                
                                messageArea.appendChild(block);
                                currentBlocks.push(block);
                            }
                        }
                    }
                    
                    // Animate blocks for this letter with precise timing
                    const letterStartTime = Date.now();
                    const targetDuration = timePerLetter * 0.8; // 80% of time for animation
                    
                    animateLetterBlocks(letterBlocks, 0, letterStartTime, targetDuration, () => {
                        // Move to next letter position
                        currentX += pattern[0].length * (blockSize + gap) - gap + letterSpacing;
                        
                        // Calculate delay for next letter to maintain total duration
                        const elapsed = Date.now() - startTime;
                        const targetElapsed = (index + 1) * timePerLetter;
                        const delay = Math.max(0, targetElapsed - elapsed);
                        
                        // Create next letter
                        setTimeout(() => createLetter(index + 1, startTime), delay);
                    });
                }
                
                // Start creating letters with timing start
                const startTime = Date.now();
                createLetter(0, startTime);
            }
            
            function animateLetterBlocks(letterBlocks, index, startTime, targetDuration, callback) {
                if (index >= letterBlocks.length) {
                    if (callback) callback();
                    return;
                }
                
                const blockData = letterBlocks[index];
                const { block, finalX, finalY, startPosX, startPosY } = blockData;
                
                // Much faster animation speeds for all messages
                const duration = 50 + Math.random() * 50; // Very fast animation
                
                function animateBlock() {
                    const elapsed = Date.now() - startTime;
                    const progress = Math.min(elapsed / duration, 1);
                    
                    // Easing function for smooth motion
                    const ease = 1 - Math.pow(1 - progress, 3);
                    
                    const currentX = startPosX + (finalX - startPosX) * ease;
                    const currentY = startPosY + (finalY - startPosY) * ease;
                    
                    block.style.left = `${currentX}px`;
                    block.style.top = `${currentY}px`;
                    block.style.opacity = progress.toString();
                    
                    if (progress < 1) {
                        requestAnimationFrame(animateBlock);
                    } else {
                        // Move to next block immediately
                        animateLetterBlocks(letterBlocks, index + 1, startTime, targetDuration, callback);
                    }
                }
                
                // Start animation immediately
                requestAnimationFrame(animateBlock);
            }

                // Create particles during explosion

                    const distance = 40 + Math.random() * 60;

                    const targetY = startY + Math.sin(angle) * distance;

                    const duration = 150 + Math.random() * 100; // Faster explosion
                    
                    function animateExplosion() {
                        const elapsed = Date.now() - startTime;
                        const progress = Math.min(elapsed / duration, 1);
                        
                        // Easing function for explosion
                        const ease = Math.pow(progress, 0.7);
                        
                        const currentX = startX + (targetX - startX) * ease;
                        const currentY = startY + (targetY - startY) * ease;
                        const rotation = progress * 360;
                        const opacity = 1 - progress;
                        
                        block.style.left = `${currentX}px`;
                        block.style.top = `${currentY}px`;
                        block.style.transform = `rotate(${rotation}deg)`;
                        block.style.opacity = opacity.toString();
                        
                        if (progress < 1) {
                            requestAnimationFrame(animateExplosion);
                        } else {
                            // Remove block after explosion
                            if (block.parentNode === messageArea) {
                                block.parentNode.removeChild(block);
                            }
                        }
                    }
                    
                    // Minimal stagger for faster explosions
                    setTimeout(() => {
                        requestAnimationFrame(animateExplosion);
                    }, index * 3); // Reduced stagger
                });
                
                // After explosion, show next message
                setTimeout(() => {
                    showNextMessage();
                }, 300); // Reduced delay
            }
